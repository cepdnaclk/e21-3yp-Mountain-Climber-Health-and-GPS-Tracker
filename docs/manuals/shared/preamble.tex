% ── Image Placeholder Command ──
\newcommand{\imagePlaceholder}[3][0.8\textwidth]{%
  \begin{figure}[H]
    \centering
    \fcolorbox{slate400}{chaptergray}{%
      \parbox{#1}{%
        \vspace{2cm}
        \centering
        \textcolor{slate400}{\large\textbf{[ IMAGE PLACEHOLDER ]}}\\[6pt]
        \textcolor{slate600}{\texttt{\detokenize{#2}}}
        \vspace{2cm}
      }%
    }
    \caption{#3}
    \label{fig:#2}
  \end{figure}
}
